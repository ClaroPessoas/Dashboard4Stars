// packages/ai-sdk 패키지 또는 직접 supermemory npm 패키지 사용
import { createClient } from "supermemory";
import { streamText } from "ai";
import { anthropic } from "@ai-sdk/anthropic";

const memory = createClient({ apiKey: process.env.SUPERMEMORY_API_KEY });

// API 라우트: /api/chat/route.ts
export async function POST(req: Request) {
  const { messages, userId } = await req.json();

  // 1. 관련 메모리 검색
  const memories = await memory.search({
    query: messages.at(-1).content,
    userId,
    limit: 5,
  });

  // 2. 메모리를 시스템 프롬프트에 주입
  const systemPrompt = `당신은 사용자의 이전 대화를 기억하는 AI입니다.
관련 기억:
${memories.map(m => `- ${m.content}`).join("\n")}`;

  // 3. LLM 호출
  const result = streamText({
    model: anthropic("claude-sonnet-4-6"),
    system: systemPrompt,
    messages,
    onFinish: async ({ text }) => {
      // 4. 대화 내용을 메모리에 저장
      await memory.add({ content: text, userId });
    },
  });

  return result.toDataStreamResponse();
}

6. 성능 및 확장성 고려사항 / Performance & Scalability

[한국어]

알려진 성능 병목 지점:

  1. 콘텐츠 인제스트 파이프라인: IngestContentWorkflow는 AI 요약, 임베딩 생성, 벡터 저장이 순차적으로 이루어지므로 대용량 파일(PDF, 동영상)은 처리 시간이 길 수 있음. Cloudflare Workflows가 비동기적으로 처리하므로 UI는 블로킹되지 않음.

  2. 벡터 검색 쿼리: 임베딩 벡터 유사도 검색은 데이터 규모에 따라 지연 시간이 증가. 결과 캐싱(Cloudflare KV)으로 반복 쿼리 최적화.

  3. 메모리 그래프 렌더링 (packages/memory-graph/): D3-force 물리 시뮬레이션은 노드 수가 많아질수록 CPU 집약적. Canvas 렌더링으로 최적화되어 있으나 1,000개 이상의 노드에서는 성능 저하 가능.

  4. Next.js 번들 크기: PDF.js, Slate, Tiptap, D3 등 무거운 라이브러리 포함. next.config.ts의 코드 스플리팅으로 초기 로드는 최적화되어 있음.

수평/수직 확장 전략:

  • 수평 확장: Cloudflare Workers는 요청당 격리된 인스턴스에서 실행되므로 자동 수평 확장. 추가 설정 불필요.
  • Durable Objects: MCP 세션은 Durable Object당 하나의 연결로 관리. 세션 수가 늘어나도 독립적으로 확장.
  • 데이터베이스 확장: Hyperdrive는 PostgreSQL 연결 풀링을 자동 처리. 쿼리 최적화는 Drizzle ORM의 인덱스 설정으로 관리.

Known Performance Bottlenecks:

  1. Content ingest pipeline: IngestContentWorkflow runs AI summarization, embedding generation, and vector storage sequentially, so large files (PDFs, videos) can take time. However, Cloudflare Workflows process asynchronously, so the UI is not blocked.

  2. Vector search queries: Embedding vector similarity search latency grows with dataset size. Caching repeated queries with Cloudflare KV mitigates this.

  3. Memory graph rendering (packages/memory-graph/): D3-force physics simulation is CPU-intensive with many nodes. Canvas rendering is used for optimization, but performance degrades beyond ~1,000 nodes.

  4. Next.js bundle size: Heavy libraries included (PDF.js, Slate, Tiptap, D3). Code splitting in next.config.ts optimizes initial load, but monitor bundle analysis on changes.

Horizontal / Vertical Scaling Strategies:

  • Horizontal scaling: Cloudflare Workers run in isolated instances per request — auto-scaling out of the box, no configuration needed.
  • Durable Objects: Each MCP session is managed by one Durable Object instance. Scales independently as session count grows.
  • Database scaling: Hyperdrive handles PostgreSQL connection pooling automatically. Query optimization is managed via Drizzle ORM index configuration.

Caching Strategy:

Layer Technology Purpose
Next.js ISR R2 (supermemory-console-cache) Static/incremental page caching
API responses Cloudflare KV Caching frequently accessed memories/documents
React Query In-memory (TanStack Query) Client-side data cache (staleTime-based)
